# The node record of graph.nodedata, storage format 3: the one definition of its layout.
#
# Every codec of the record is tested against this file, so a field added to one of them and
# not to the others fails in CI: the JVM writer and reader (NodeRecordCodec.kt), the JVM build
# spill (MmapGraphBuilder.kt, same fields, its own string encoding), the Rust reader
# (node.rs), the Rust writer (interchange.rs) and the readers at fixed byte offsets
# (columns.rs RAW_STRING_FIELDS). See NodeRecordSchemaTest.kt and node_schema_tests.rs.
#
# A record is: id:i32 tag:u8, then the kind's fields in order. Everything is big-endian.
#
#   kind <tag> <Kind> <field>:<encoding> ...
#
# Encodings:
#   i32 i64 f32 f64   fixed width
#   bool              one byte, 0 or 1
#   sid               i32 index into graph.strings
#   sid?              bool; a sid follows when it is 1
#   i32?              i32, -1 when absent
#   ids               i32 count, then that many i32 node ids
#   method            class:sid name:sid, i32 count and that many parameter-type sids, return:sid
#   value             one tagged value (the `value` lines below)
#   values            i32 count, then that many tagged values
#   pairs             i32 count, then that many (sid, value)
#
# Not in the record: a call site's ordinal and origin (graph.callsite-ordinals), type
# arguments of a type (graph.types), and a method's node identity.

kind 0 IntConstant value:i32
kind 1 StringConstant value:sid
kind 2 LongConstant value:i64
kind 3 FloatConstant value:f32
kind 4 DoubleConstant value:f64
kind 5 BooleanConstant value:bool
kind 6 NullConstant
kind 7 EnumConstant enum_type:sid name:sid args:values
kind 8 LocalVariable name:sid type:sid method:method
kind 9 Field class:sid name:sid type:sid is_static:bool
kind 10 Parameter index:i32 type:sid method:method
kind 11 Return method:method actual_type:sid?
kind 12 CallSite caller:method callee:method line:i32? receiver:i32? arguments:ids
kind 13 Annotation name:sid class:sid member:sid values:pairs
kind 14 ResourceValue path:sid key:sid value:value format:sid profile:sid?
kind 15 ResourceFile path:sid source:sid format:sid profile:sid?
kind 16 ClassConstant value:sid
